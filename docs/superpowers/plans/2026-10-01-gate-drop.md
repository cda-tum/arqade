# Gate Drop Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build Gate Drop, a two-player Dr. Mario-style versus puzzle game where
gate pills fall onto qubit wires, identical gates cancel, and the first player
to clear every Pauli error on their board wins. Then ship it to the cabinet.

**Architecture:** A new uv project `gate-drop/`, shaped like the sibling games.
`board.py` (one player's board: pills, settling, cancellation, chains),
`levels.py` (error layouts) and `match.py` (the match flow and the attack
exchange) are pure Python and carry most of the tests. `sprites.py` and
`render.py` draw SNES-style pixel art at 480x270. `audio.py` is chiptune
adapted from Overquqed's.

**Tech Stack:** Python 3.11, pygame 2.5.2, pytest, uv.

**Spec:** `docs/superpowers/specs/2026-10-01-gate-drop-design.md`

## Global Constraints

- `requires-python = ">=3.11,<3.12"` and `pygame==2.5.2` exactly.
- LF line endings in every new file. `deploy_batocera.sh` is committed executable.
- Raw buttons: A = 1 (rotate clockwise in game, confirm in menus), B = 2 (rotate
  counter-clockwise in game, back in menus), Select = 8, Start = 9.

- The cabinet loader calls `main(["--fullscreen", "--no-keyboard"])`.

- P1 (left) is blue and P2 (right) red (AGENTS.md "Player colours").
- `board.py`, `levels.py` and `match.py` never import pygame.

```python
"""Fixed internal-resolution rendering, scaled onto the real display.

Fullscreen (the cabinet) uses pygame's SCALED mode, so SDL does the x4
upscale on the GPU (nearest-neighbour, set in main.py). Windowed dev mode
scales and blits manually, because pygame 2.5.2's SCALED mode has no way to
pick the window size. Same approach as the sibling games.
"""

from __future__ import annotations

import pygame

from gate_drop.config import INTERNAL_HEIGHT, INTERNAL_WIDTH


def compute_scaled_rect(
    internal_size: tuple[int, int], window_size: tuple[int, int]
) -> pygame.Rect:
    """Largest rect of internal_size's aspect ratio that fits in window_size, centered."""
    internal_w, internal_h = internal_size
    window_w, window_h = window_size
    scale = min(window_w / internal_w, window_h / internal_h)
    scaled_w = round(internal_w * scale)
    scaled_h = round(internal_h * scale)
    return pygame.Rect((window_w - scaled_w) // 2, (window_h - scaled_h) // 2, scaled_w, scaled_h)


def make_display(fullscreen: bool, window_size: str) -> tuple[pygame.Surface, pygame.Surface | None]:
    """Returns (surface_to_draw_on, window_or_None). In fullscreen the surface
    is what gets presented, so window is None and present() is a no-op."""
    if fullscreen:
        surface = pygame.display.set_mode(
            (INTERNAL_WIDTH, INTERNAL_HEIGHT), pygame.SCALED | pygame.FULLSCREEN
        )
        return surface, None
    width_str, height_str = window_size.lower().split("x")
    window = pygame.display.set_mode((int(width_str), int(height_str)), pygame.RESIZABLE)
    return pygame.Surface((INTERNAL_WIDTH, INTERNAL_HEIGHT)), window


def present(internal_surface: pygame.Surface, window: pygame.Surface | None) -> None:
    """Scale internal_surface onto window, letterboxed. No-op in fullscreen."""
    if window is None:
        return
    rect = compute_scaled_rect(internal_surface.get_size(), window.get_size())
    window.fill((0, 0, 0))
    window.blit(pygame.transform.scale(internal_surface, rect.size), rect)
```

```python
import random

from gate_drop import config
from gate_drop.board import (
    Board, Cnot, Gate, Grid, Piece, PlayerInput, apply_matches, find_matches, settle_step,
)

B = config.ROWS - 1  # the bottom row


def grid_from(*rows):
    """The bottom rows of a board, listed top to bottom: '.' empty, H X Y Z
    gates, x y z errors, c and t a CNOT's control and target (each paired with
    the t or c right next to it)."""
    g = Grid()
    top = config.ROWS - len(rows)
    pill = 100
    for dy, row in enumerate(rows):
        y, x = top + dy, 0
        while x < len(row):
            ch = row[x]
            if ch in "HXYZ":
                g.set((x, y), Gate(ch))
            elif ch in "xyz":
                g.set((x, y), Gate(ch.upper(), error=True))
            elif ch in "ct":
                g.set((x, y), Cnot(pill, ch))
                g.set((x + 1, y), Cnot(pill, row[x + 1]))
                pill += 1
                x += 1
            x += 1
    return g


def column(g, x):
    """A wire's contents from the bottom up, as letters (errors lower-case)."""
    out = []
    for y in range(config.ROWS - 1, -1, -1):
        c = g.get((x, y))
        if isinstance(c, Gate):
            out.append(c.kind.lower() if c.error else c.kind)
        elif isinstance(c, Cnot):
            out.append(c.role)
    return "".join(out)


def settle_all(g):
    while settle_step(g):
        pass


def resolve(g):
    """Run every match pass to the end, settling between, as the board does."""
    passes = 0
    settle_all(g)
    while (matches := find_matches(g)):
        apply_matches(g, matches)
        settle_all(g)
        passes += 1
    return passes


# --- what cancels -------------------------------------------------------------------------


def test_two_identical_gates_on_a_wire_cancel():
    for kind in "HXYZ":
        g = grid_from(kind + ".....", kind + ".....")
        assert [m.kind for m in find_matches(g)] == ["pair"]
        resolve(g)
        assert column(g, 0) == ""


def test_different_gates_or_different_wires_do_not_cancel():
    assert find_matches(grid_from("X.....", "Z.....")) == []
    assert find_matches(grid_from("XX....")) == []


def test_gaps_on_a_wire_are_skipped():
    assert [m.kind for m in find_matches(grid_from("X.....", "......", "X....."))] == ["pair"]


def test_a_gate_clears_a_matching_error():
    g = grid_from("Y.....", "y.....")
    changed, errors = apply_matches(g, find_matches(g))
    assert errors == 1 and column(g, 0) == ""


def test_an_h_sandwich_turns_x_into_z_and_z_into_x():
    g = grid_from("H.....", "X.....", "H.....")
    resolve(g)
    assert column(g, 0) == "Z"
    g = grid_from("H.....", "Z.....", "H.....")
    resolve(g)
    assert column(g, 0) == "X"


def test_h_y_h_changes_nothing():
    assert find_matches(grid_from("H.....", "Y.....", "H.....")) == []


def test_a_sandwiched_error_stays_an_error():
    g = grid_from("H.....", "x.....", "H.....")
    _, errors = apply_matches(g, find_matches(g))
    assert errors == 0
    assert column(g, 0) == "z"


def test_wires_are_scanned_from_the_bottom_without_overlap():
    g = grid_from("X.....", "X.....", "X.....")
    matches = find_matches(g)
    assert len(matches) == 1 and matches[0].removed == ((0, B), (0, B - 1))
    g = grid_from("H.....", "H.....", "X.....", "H.....")  # bottom up: H X H H
    resolve(g)
    assert column(g, 0) == "ZH"


def test_nothing_cancels_across_a_cnot():
    assert find_matches(grid_from("X.....", "ct....", "X.....")) == []


def test_two_stacked_cnots_cancel():

    assert [m.kind for m in find_matches(g)] == ["cnot"]

    assert find_matches(grid_from("ct....", ".X....", "ct....")) == []

def test_a_loose_gate_falls_one_row_per_step():
    g = grid_from("X.....", "......", "......")
    assert settle_step(g) == [((0, B - 2), (0, B - 1))]
    assert settle_step(g) == [((0, B - 1), (0, B))]

    assert g.get((1, B)) is None


def test_halves_that_end_up_apart_come_unjoined():
    g = Grid()
    g.set((0, B - 2), Gate("X", pill=7))
    g.set((1, B - 2), Gate("Z", pill=7))

    g.set((0, B), Gate("Z"))
    settle_all(g)
    assert g.get((0, B - 2)) == Gate("X")
    assert g.get((1, B)) == Gate("Z")

def make_board(errors=(), piece=None):
    b = Board(list(errors), seed=1)
    b.update(0.0, PlayerInput())  # spawns the first pill
    if piece is not None:
        b.piece = piece
    return b


def run(b, seconds, inp=PlayerInput(), dt=1 / 60):
    events = []
    for _ in range(round(seconds / dt)):
        events += b.update(dt, inp)
    return events


def drop(b):
    """Soft-drop the current pill, then let everything resolve until the next
    pill is falling (without touching it)."""
    events = []
    for _ in range(2000):
        if any(e.kind == "land" for e in events):
            break
        events += b.update(1 / 60, PlayerInput(soft=True))
    for _ in range(2000):
        if b.phase in ("fall", "over"):
            break
        events += b.update(1 / 60, PlayerInput())
    return events


def test_the_first_pill_spawns_lying_flat_over_the_middle_wires():
    b = make_board()
    assert b.phase == "fall"
    assert [pos for pos, _ in b.piece.cells()] == [(2, 0), (3, 0)]


def test_moving_stops_at_the_walls():
    b = make_board()
    for _ in range(5):
        b.update(0.0, PlayerInput(move=-1))
    assert b.piece.x == 0
    for _ in range(9):
        b.update(0.0, PlayerInput(move=1))
    assert b.piece.x == config.COLS - 2


def test_rotating_cycles_through_four_orientations():
    piece = Piece((Gate("X", pill=9), Gate("Z", pill=9)), 2, 5)
    b = make_board(piece=piece)
    seen = []
    for _ in range(4):
        b.update(0.0, PlayerInput(cw=True))
        seen.append([pos for pos, _ in b.piece.cells()])
    assert seen == [[(2, 4), (2, 5)], [(3, 5), (2, 5)], [(2, 5), (2, 4)], [(2, 5), (3, 5)]]


def test_rotating_against_the_right_wall_kicks_left():
    piece = Piece((Gate("X", pill=9), Gate("Z", pill=9)), config.COLS - 1, 5, rot=1)
    b = make_board(piece=piece)
    b.update(0.0, PlayerInput(cw=True))
    assert (b.piece.x, b.piece.rot) == (config.COLS - 2, 2)


def test_a_cnot_only_swaps_its_sides():
    piece = Piece((Cnot(9, "c"), Cnot(9, "t")), 2, 5)
    b = make_board(piece=piece)
    b.update(0.0, PlayerInput(cw=True))
    assert [half.role for _, half in sorted(b.piece.cells())] == ["t", "c"]


def test_gravity_and_soft_drop():
    b = make_board()
    run(b, config.FALL_START + 0.01)
    assert b.piece.y == 1
    run(b, 0.2, PlayerInput(soft=True))
    assert b.piece.y >= 4


def test_a_landed_pill_locks_and_the_next_one_spawns():
    b = make_board(piece=Piece((Gate("Y", pill=9), Gate("Z", pill=9)), 2, 0))
    events = drop(b)
```

```python
"""One player's board: a small quantum circuit with gate pills falling onto it.

Pure Python, no pygame, so every rule is unit tested (tests/test_board.py).

The board is COLS wires (columns) by ROWS rows, with row 0 at the top. Each
column is one qubit wire, and the gates stacked on it are that qubit's
circuit. Cancellation only ever happens along a wire. Each wire is read from
the bottom up, skipping empty cells, because an idle stretch of wire is the
identity. A CNOT half is an item too, and nothing cancels across it, since
there is no commuting.

Board.update() runs the falling pill and, after it locks, the resolve
sequence: settle everything unsupported a row at a time, find the matches,
flash them, clear them, settle again, and so on until nothing matches. Then
any incoming noise drops, and the next pill spawns.
"""

from __future__ import annotations

import random
from dataclasses import dataclass

from gate_drop import config

Pos = tuple[int, int]


@dataclass(frozen=True)
class Gate:
    """A single-qubit gate. error=True marks one of the round's Pauli errors,
    anchored in place. pill is set while the gate is still joined to the
    other half of the pill it came in (for drawing them as one capsule)."""

    kind: str
    error: bool = False
    pill: int | None = None


@dataclass(frozen=True)
class Cnot:
    """One half of a CNOT: role "c" (control) or "t" (target). The two halves
    share a pill id and always sit side by side, at the same height."""

    pill: int
    role: str


# Where a pill's two halves sit relative to its pivot (its lower-left cell)
# in each rotation. Turning clockwise takes the first half from the left to
# the top, to the right, to the bottom. A CNOT only uses 0 and 2.
ROTATIONS = {0: ((0, 0), (1, 0)), 1: ((0, -1), (0, 0)), 2: ((1, 0), (0, 0)), 3: ((0, 0), (0, -1))}
SANDWICH = {"X": "Z", "Z": "X"}  # H X H = Z and H Z H = X


@dataclass
class Piece:
    halves: tuple  # (first half, second half): two Gates or two Cnots
    x: int
    y: int
    rot: int = 0

    @property
    def is_cnot(self) -> bool:
        return isinstance(self.halves[0], Cnot)

    def cells(self, x: int | None = None, y: int | None = None, rot: int | None = None) -> list:
        x = self.x if x is None else x
        y = self.y if y is None else y
        offsets = ROTATIONS[self.rot if rot is None else rot]
        return [((x + dx, y + dy), half) for (dx, dy), half in zip(offsets, self.halves)]


@dataclass(frozen=True)
class PlayerInput:
    move: int = 0  # -1 left, 1 right (already auto-repeated)
    cw: bool = False
    ccw: bool = False
    soft: bool = False  # held: soft drop


@dataclass(frozen=True)
class Event:
    kind: str
    cells: tuple[Pos, ...] = ()
    value: int = 0


@dataclass(frozen=True)
class Match:
    kind: str  # "pair", "sandwich" or "cnot"
    removed: tuple[Pos, ...]
    rewrite: tuple[Pos, str] | None = None  # the sandwiched gate and its new kind

    @property
    def cells(self) -> tuple[Pos, ...]:
        return self.removed + ((self.rewrite[0],) if self.rewrite else ())


class Grid:
    def __init__(self, cols: int = config.COLS, rows: int = config.ROWS) -> None:
        self.cols, self.rows = cols, rows
        self.cells: list[list] = [[None] * cols for _ in range(rows)]

    def get(self, p: Pos):
        x, y = p
        if 0 <= x < self.cols and 0 <= y < self.rows:
            return self.cells[y][x]
        return None

    def set(self, p: Pos, value) -> None:
        self.cells[p[1]][p[0]] = value

    def free(self, p: Pos) -> bool:
        """Can a falling pill occupy p? The rows above the top are free (the
        spawn chute); the side walls and the floor are not."""
        x, y = p
        if not 0 <= x < self.cols or y >= self.rows:
            return False
        return y < 0 or self.cells[y][x] is None

    def items(self):
        for y in range(self.rows):
            for x in range(self.cols):
                c = self.cells[y][x]
                if c is not None:
                    yield (x, y), c


def random_piece(rng: random.Random, pill: int) -> Piece:
    if rng.random() < config.CNOT_SHARE:
        halves = (Cnot(pill, "c"), Cnot(pill, "t"))
    else:
        halves = (Gate(rng.choice(config.GATE_KINDS), pill=pill), Gate(rng.choice(config.GATE_KINDS), pill=pill))
    return Piece(halves, config.SPAWN_X, 0)


# --- cancellation ----------------------------------------------------------------------


def _wire(grid: Grid, x: int) -> list:
    """A wire's items from the bottom up, skipping gaps: [(y, cell), ...]."""
    return [(y, grid.cells[y][x]) for y in range(grid.rows - 1, -1, -1) if grid.cells[y][x] is not None]


def _cnot_pair(grid: Grid, x: int, lower: tuple, upper: tuple) -> Match | None:
    """Two CNOT halves directly stacked on wire x: do the whole CNOTs cancel?"""
    (y0, c0), (y1, c1) = lower, upper
    if not (isinstance(c0, Cnot) and isinstance(c1, Cnot)) or c0.pill == c1.pill or c0.role != c1.role:
        return None
    for dx in (1, -1):
        p0, p1 = grid.get((x + dx, y0)), grid.get((x + dx, y1))
        if isinstance(p0, Cnot) and p0.pill == c0.pill and isinstance(p1, Cnot) and p1.pill == c1.pill:
            if all(grid.get((x + dx, y)) is None for y in range(y1 + 1, y0)):
                return Match("cnot", ((x, y0), (x + dx, y0), (x, y1), (x + dx, y1)))
            return None
    return None


def find_matches(grid: Grid) -> list[Match]:
    """Every match on the board, scanning each wire from the bottom: at each
    position a sandwich first, then a pair. Matches never overlap."""
    matches: list[Match] = []
    taken: set[Pos] = set()
    for x in range(grid.cols):
        seq = _wire(grid, x)
        i = 0
        while i < len(seq):
            y0, c0 = seq[i]
            if (x, y0) in taken:
                i += 1
                continue
            if i + 2 < len(seq):
                (y1, c1), (y2, c2) = seq[i + 1], seq[i + 2]
                if (_is_h(c0) and _is_h(c2) and isinstance(c1, Gate) and c1.kind in SANDWICH
                        and not {(x, y1), (x, y2)} & taken):
                    matches.append(Match("sandwich", ((x, y0), (x, y2)), ((x, y1), SANDWICH[c1.kind])))
                    taken |= {(x, y0), (x, y1), (x, y2)}
                    i += 3
                    continue
            if i + 1 < len(seq):
                y1, c1 = seq[i + 1]
                if (x, y1) not in taken:
                    if isinstance(c0, Gate) and isinstance(c1, Gate) and c0.kind == c1.kind:
                        matches.append(Match("pair", ((x, y0), (x, y1))))
                        taken |= {(x, y0), (x, y1)}
                        i += 2
                        continue
                    m = _cnot_pair(grid, x, seq[i], seq[i + 1])
                    if m is not None and not set(m.removed) & taken:
                        matches.append(m)
                        taken |= set(m.removed)
                        i += 2
                        continue
            i += 1
    return matches


def _is_h(c) -> bool:
    return isinstance(c, Gate) and c.kind == "H"


def apply_matches(grid: Grid, matches: list[Match]) -> tuple[list[Pos], int]:
    """Clear the matches. Returns (every cell that changed, errors cleared)."""
    changed: list[Pos] = []
    errors = 0
    for m in matches:
        for p in m.removed:
            c = grid.get(p)
            if isinstance(c, Gate) and c.error:
                errors += 1
            grid.set(p, None)
            changed.append(p)
        if m.rewrite is not None:
            p, kind = m.rewrite
            old = grid.get(p)
            grid.set(p, Gate(kind, error=old.error))
            changed.append(p)
    _fix_joins(grid)
    return changed, errors


# --- settling -------------------------------------------------------------------------


def settle_step(grid: Grid) -> list[tuple[Pos, Pos]]:
    """Move everything unsupported down one row. Rows are processed from the
    bottom up, so a whole stack falls together. Loose gates fall alone, a
    CNOT falls only when both of its halves can, and errors never move."""
    moves: list[tuple[Pos, Pos]] = []
    for y in range(grid.rows - 2, -1, -1):
        for x in range(grid.cols):
            c = grid.cells[y][x]
            if c is None:
                continue
            if isinstance(c, Gate):
                if not c.error and grid.cells[y + 1][x] is None:
                    grid.cells[y + 1][x], grid.cells[y][x] = c, None
                    moves.append(((x, y), (x, y + 1)))
            elif x + 1 < grid.cols and isinstance(grid.cells[y][x + 1], Cnot) and grid.cells[y][x + 1].pill == c.pill:
                if grid.cells[y + 1][x] is None and grid.cells[y + 1][x + 1] is None:
                    grid.cells[y + 1][x], grid.cells[y][x] = c, None
                    grid.cells[y + 1][x + 1], grid.cells[y][x + 1] = grid.cells[y][x + 1], None
                    moves += [((x, y), (x, y + 1)), ((x + 1, y), (x + 1, y + 1))]
    _fix_joins(grid)
    return moves


def _fix_joins(grid: Grid) -> None:
    """A pill half whose partner is gone, or no longer next to it, is loose."""
    for (x, y), c in list(grid.items()):
        if isinstance(c, Gate) and c.pill is not None:
            near = (grid.get(q) for q in ((x + 1, y), (x - 1, y), (x, y + 1), (x, y - 1)))
            if not any(isinstance(n, Gate) and n.pill == c.pill for n in near):
                grid.set((x, y), Gate(c.kind, c.error))


# --- the board --------------------------------------------------------------------------


class Board:
    def __init__(self, errors: list[tuple[Pos, str]], seed: int) -> None:
        self.grid = Grid()
        for pos, kind in errors:
            self.grid.set(pos, Gate(kind, error=True))
        self.rng = random.Random(seed)  # pills only, so both boards get the same ones
        self.noise_rng = random.Random(seed + 1)
        self._pill = 0
        self.piece: Piece | None = None
        self.next = self._new_piece()
        self.phase = "spawn"  # spawn, fall, settle, flash or over
        self.timer = 0.0
        self.fall_timer = 0.0
        self.fall_interval = config.FALL_START
        self.chain = 0
        self.big_pass = False
        self.pending: list[Match] = []
        self.noise_queue: list[str] = []
        self.noise_dropped = False
        self.outgoing = 0  # noise gates to send to the opponent (the match collects them)
        self.topped_out = False

    def _new_piece(self) -> Piece:
        self._pill += 1
        return random_piece(self.rng, self._pill)

    def errors_left(self) -> int:
        return sum(1 for _, c in self.grid.items() if isinstance(c, Gate) and c.error)

    def stack_top(self) -> int:
        """The highest occupied row (ROWS if the board is empty)."""
        for y in range(self.grid.rows):
            if any(c is not None for c in self.grid.cells[y]):
                return y
        return self.grid.rows

    def update(self, dt: float, inp: PlayerInput = PlayerInput()) -> list[Event]:
        events: list[Event] = []
        if self.topped_out:
            return events
        if self.phase == "spawn":
            self._spawn(events)
        elif self.phase == "fall":
            self._fall(dt, inp, events)
        elif self.phase == "flash":
            self.timer -= dt
            if self.timer <= 0:
                self._clear(events)
        elif self.phase == "settle":
            self.timer -= dt
            while self.phase == "settle" and self.timer <= 0:
                self._resolve_step(events)
        return events

    # --- the falling pill ---------------------------------------------------------------

    def _fits(self, piece: Piece, x: int | None = None, y: int | None = None, rot: int | None = None) -> bool:
        return all(self.grid.free(p) for p, _ in piece.cells(x, y, rot))

    def _spawn(self, events: list[Event]) -> None:
        self.piece, self.next = self.next, self._new_piece()
        if not self._fits(self.piece):
            self.piece = None
            self._top_out(events)
            return
        self.phase = "fall"
        self.fall_timer = 0.0
        events.append(Event("spawn"))

    def _fall(self, dt: float, inp: PlayerInput, events: list[Event]) -> None:
        piece = self.piece
        if inp.move and self._fits(piece, x=piece.x + inp.move):
            piece.x += inp.move
            events.append(Event("move"))
        for turn, pressed in ((1, inp.cw), (-1, inp.ccw)):
            if pressed:
                self._rotate(turn, events)
        interval = config.SOFT_DROP_INTERVAL if inp.soft else self.fall_interval
        self.fall_timer += dt
        while self.fall_timer >= interval:
            self.fall_timer -= interval
            if self._fits(piece, y=piece.y + 1):
                piece.y += 1
            else:
                self._lock(events)
                return

    def _rotate(self, turn: int, events: list[Event]) -> None:
        piece = self.piece
        rot = (piece.rot + 2) % 4 if piece.is_cnot else (piece.rot + turn) % 4
        for x in (piece.x, piece.x - 1):  # in place, else a one-cell wall kick to the left
            if self._fits(piece, x=x, rot=rot):
                piece.x, piece.rot = x, rot
                events.append(Event("rotate"))
                return

    def _lock(self, events: list[Event]) -> None:
        cells = self.piece.cells()
        self.piece = None
        if any(y < 0 for (_, y), _ in cells):
            self._top_out(events)
            return
        for pos, half in cells:
            self.grid.set(pos, half)
        events.append(Event("land", tuple(pos for pos, _ in cells)))
        self.phase = "settle"
        self.timer = 0.0
        self.chain = 0
        self.big_pass = False

    def _top_out(self, events: list[Event]) -> None:
        self.topped_out = True
        self.phase = "over"
        events.append(Event("topout"))

    # --- resolving ----------------------------------------------------------------------

    def _resolve_step(self, events: list[Event]) -> None:
        moves = settle_step(self.grid)
        if moves:
            events.append(Event("settle", tuple(to for _, to in moves)))
            self.timer += config.SETTLE_STEP
            return
        matches = find_matches(self.grid)
        if matches:
            self.chain += 1
            self.big_pass |= len(matches) >= config.BIG_PASS
            self.pending = matches
            self.phase = "flash"
            self.timer = config.CLEAR_TIME
            events.append(Event("flash", tuple(c for m in matches for c in m.cells), self.chain))
            return
        if self.chain:
            sent = min(config.MAX_ATTACK, self.chain - 1) + (1 if self.big_pass else 0)
            events.append(Event("chain", value=self.chain))
            if sent:
                self.outgoing += sent
                events.append(Event("attack", value=sent))
            self.chain = 0
            self.big_pass = False
        if self.noise_queue and not self.noise_dropped:
            self.noise_dropped = True
            self._drop_noise(events)
            self.timer += config.SETTLE_STEP
            return
        self.noise_dropped = False
        self.phase = "spawn"
        self._spawn(events)

    def _clear(self, events: list[Event]) -> None:
        changed, errors = apply_matches(self.grid, self.pending)
        self.pending = []
        events.append(Event("clear", tuple(changed), self.chain))
        if errors:
            events.append(Event("errors", value=errors))
        self.phase = "settle"
        self.timer = config.SETTLE_STEP

    def _drop_noise(self, events: list[Event]) -> None:
        """Up to one noise gate per wire, into the top row, on distinct wires."""
        count = min(len(self.noise_queue), self.grid.cols)
        landed = []
        for x in self.noise_rng.sample(range(self.grid.cols), count):
            kind = self.noise_queue.pop(0)
            if self.grid.get((x, 0)) is None:
                self.grid.set((x, 0), Gate(kind))
                landed.append((x, 0))
        events.append(Event("noise", tuple(landed)))
```

```python
"""Error layouts: where each round's Pauli errors sit. Pure and seeded, so
both boards get exactly the same layout."""

from __future__ import annotations

import random

from gate_drop import config


def errors_for_round(n: int) -> int:
    return min(config.ERRORS_MAX, config.ERRORS_FIRST + config.ERRORS_STEP * (n - 1))


def error_layout(rng: random.Random, count: int) -> list:
    """`count` errors on distinct cells in the bottom ERROR_ROWS rows. Two
    identical errors are never next to each other on a wire (gaps skipped),
    so none cancels out on its own."""
    top = config.ROWS - config.ERROR_ROWS
    cells = rng.sample([(x, y) for x in range(config.COLS) for y in range(top, config.ROWS)], count)
    layout = []
    for x in range(config.COLS):
        below = None
        for pos in sorted((c for c in cells if c[0] == x), key=lambda c: -c[1]):
            kind = rng.choice([k for k in config.PAULIS if k != below])
            layout.append((pos, kind))
            below = kind
    return sorted(layout)
```

```python
"""Match flow: title -> countdown -> round -> round result -> ... -> match
result. Pure (no pygame): it takes one FrameInput per frame and returns
events, so it is unit tested like board.py.

Both boards get the same error layout and the same pill sequence each round.
Chains on one board send noise gates to the other. A round ends when a
board has no errors left, or when a board tops out.
"""

from __future__ import annotations

import random
from dataclasses import dataclass, field

from gate_drop import config, levels
from gate_drop.board import Board, Event, PlayerInput

PLAYERS = ("p1", "p2")
TITLE = "title"
COUNTDOWN = "countdown"
ROUND = "round"
ROUND_RESULT = "round_result"
MATCH_RESULT = "match_result"


@dataclass(frozen=True)
class Cue:
    """A match-level event for the sound and the renderer."""

    kind: str  # "beep", "go", "round_won", "draw" or "match_won"
    player: str | None = None


@dataclass(frozen=True)
class BoardEvent:
    player: str
    event: Event


def _idle() -> dict:
    return {p: PlayerInput() for p in PLAYERS}


@dataclass
class FrameInput:
    players: dict = field(default_factory=_idle)  # player -> board.PlayerInput
    confirm: bool = False  # A or Start pressed this frame, either pad
    back: bool = False  # B pressed this frame, either pad
    activity: bool = False  # any input at all (for the idle timeout)


def other(player: str) -> str:
    return "p2" if player == "p1" else "p1"


def fall_interval(t: float) -> float:
    """Seconds per row, t seconds into a round."""
    steps = int(t // config.FALL_SPEEDUP_EVERY)
    return max(config.FALL_MIN, config.FALL_START * config.FALL_SPEEDUP ** steps)


class Match:
    def __init__(self, rng: random.Random | None = None) -> None:
        self.rng = rng or random.Random()
        self.state = TITLE
        self.t = 0.0  # seconds in the current state
        self.idle = 0.0
        self.scores = {p: 0 for p in PLAYERS}
        self.boards: dict[str, Board] = {}
        self.round_no = 1
        self.round_time = 0.0
        self.last_winner: str | None = None
        self.match_winner: str | None = None
        self._beeps = 0

    @property
    def danger(self) -> bool:
        """A stack is close to the top (the music speeds up, the frame blinks)."""
        return self.state == ROUND and any(b.stack_top() < config.DANGER_ROWS for b in self.boards.values())

    def update(self, dt: float, inp: FrameInput) -> list:
        self.idle = 0.0 if inp.activity else self.idle + dt
        if self.state != TITLE and self.idle >= config.IDLE_TIMEOUT:
            self.boards = {}
            self._set_state(TITLE)
            return []
        self.t += dt
        events: list = []
        if self.state == TITLE:
            if inp.confirm:
                self._start_match(events)
        elif self.state == COUNTDOWN:
            if self.t >= config.COUNTDOWN:
                self._set_state(ROUND)
                self.round_time = 0.0
                events.append(Cue("go"))
            elif int(self.t) >= self._beeps:
                self._beeps += 1
                events.append(Cue("beep"))
        elif self.state == ROUND:
            self._update_round(dt, inp, events)
        elif self.state == ROUND_RESULT:
            if self.t >= config.ROUND_RESULT_TIME:
                leader = max(self.scores, key=self.scores.get)
                if self.scores[leader] >= config.ROUNDS_TO_WIN:
                    self.match_winner = leader
                    self._set_state(MATCH_RESULT)
                    events.append(Cue("match_won", leader))
                else:
                    if self.last_winner is not None:  # a draw replays the same round
                        self.round_no += 1
                    self._start_countdown(events)
        elif self.state == MATCH_RESULT:
            if self.t >= config.MATCH_RESULT_LOCK:
                if inp.confirm:
                    self._start_match(events)
                elif inp.back:
                    self.boards = {}
                    self._set_state(TITLE)
        return events

    def _set_state(self, state: str) -> None:
        self.state = state
        self.t = 0.0

    def _start_match(self, events: list) -> None:
        self.scores = {p: 0 for p in PLAYERS}
        self.match_winner = None
        self.round_no = 1
        self._start_countdown(events)

    def _start_countdown(self, events: list) -> None:
        seed = self.rng.randrange(1 << 30)
        layout = levels.error_layout(random.Random(seed), levels.errors_for_round(self.round_no))
        self.boards = {p: Board(layout, seed) for p in PLAYERS}
        self.round_time = 0.0
        self._set_state(COUNTDOWN)
        self._beeps = 1
        events.append(Cue("beep"))

    def _update_round(self, dt: float, inp: FrameInput, events: list) -> None:
        self.round_time += dt
        interval = fall_interval(self.round_time)
        for p in PLAYERS:
            board = self.boards[p]
            board.fall_interval = interval
            events += [BoardEvent(p, e) for e in board.update(dt, inp.players.get(p, PlayerInput()))]
        for p in PLAYERS:
            board = self.boards[p]
            if board.outgoing:
                self.boards[other(p)].noise_queue += [self.rng.choice(config.PAULIS) for _ in range(board.outgoing)]
                board.outgoing = 0
        wins = {p for p in PLAYERS if self.boards[p].errors_left() == 0 or self.boards[other(p)].topped_out}
        if not wins:
            return
        if len(wins) == 2:
            self.last_winner = None
            events.append(Cue("draw"))
        else:
            winner = wins.pop()
            self.last_winner = winner
            self.scores[winner] += 1
            events.append(Cue("round_won", winner))
        self._set_state(ROUND_RESULT)
```

```python
"""Pads and the dev keyboard, polled once per frame into a match.FrameInput.

Per player: the stick moves the pill left and right (auto-repeated) and
soft-drops it when pushed down; A rotates clockwise and B anti-clockwise.
In menus, A (or Start) confirms and B goes back. The pure helpers are unit
tested; reading real pads is verified on the cabinet with --input-test.
"""

from __future__ import annotations

import pygame

from gate_drop import config
from gate_drop.board import PlayerInput
from gate_drop.match import PLAYERS, FrameInput

# Batocera's evmapy turns P1's pad into keystrokes too (see ../../../AGENTS.md),
# so the cabinet loader passes --no-keyboard and only the pads are read.
_keyboard_enabled = True


def set_keyboard_enabled(enabled: bool) -> None:
    global _keyboard_enabled
    _keyboard_enabled = enabled


class _NoKeys:
    """Stands in for pygame.key.get_pressed() while the keyboard is off."""

    def __getitem__(self, key: int) -> bool:
        return False


def read_keys():
    return pygame.key.get_pressed() if _keyboard_enabled else _NoKeys()


def _codes(names: list[str]) -> list[int]:
    return [getattr(pygame, name) for name in names]


def keyboard_quit_held(keys) -> bool:
    return bool(keys[getattr(pygame, config.INPUT_CONFIG["system"]["keyboard_quit"])])


def axis_sign(value: float, deadzone: float) -> int:
    if value < -deadzone:
        return -1
    if value > deadzone:
        return 1
    return 0


def should_quit(*, p1_start: bool, p1_select: bool, p2_start: bool, p2_select: bool) -> bool:
    """Start + Select held together on the same pad."""
    return (p1_start and p1_select) or (p2_start and p2_select)


class AutoRepeat:
    """Turns a held direction (-1, 0 or 1) into presses: one at once, then
    one after REPEAT_DELAY, then one every REPEAT_INTERVAL."""

    def __init__(self, delay: float = config.REPEAT_DELAY, interval: float = config.REPEAT_INTERVAL) -> None:
        self.delay, self.interval = delay, interval
        self.held = 0
        self.timer = 0.0

    def update(self, dt: float, value: int) -> int:
        if value == 0:
            self.held = 0
            return 0
        if value != self.held:
            self.held = value
            self.timer = self.delay
            return value
        self.timer -= dt
        if self.timer <= 0:
            self.timer += self.interval
            return value
        return 0


class Input:
    def __init__(self) -> None:
        self.joysticks: list = []
        self.keymap = {p: {act: _codes(names) for act, names in config.KEYMAP[p].items()} for p in PLAYERS}
        self.confirm_keys = _codes(config.KEYS_CONFIRM)
        self.back_keys = _codes(config.KEYS_BACK)
        self.repeat = {p: AutoRepeat() for p in PLAYERS}
        self._down: dict[str, set[int]] = {p: set() for p in PLAYERS}
        self._keys_prev: dict[str, set[str]] = {p: set() for p in PLAYERS}
        self._sys_prev: set[str] = set()
        self.inputs = {p: PlayerInput() for p in PLAYERS}
        self.confirm = self.back = self.activity = False
        self.start_select = {p: (False, False) for p in PLAYERS}
        self.select_held_time = 0.0
        self.keyboard_quit = False
        self.rescan()

    def rescan(self) -> None:
        self.joysticks = []
        for i in range(pygame.joystick.get_count()):
            joy = pygame.joystick.Joystick(i)
            joy.init()
            self.joysticks.append(joy)

    def handle(self, event: pygame.event.Event) -> None:
        if event.type in (pygame.JOYDEVICEADDED, pygame.JOYDEVICEREMOVED):
            self.rescan()

    def _read_pad(self, player: str) -> tuple[set[int], int, bool]:
        cfg = config.INPUT_CONFIG[player]
        if cfg["joystick_index"] >= len(self.joysticks):
            return set(), 0, False
        joy = self.joysticks[cfg["joystick_index"]]
        try:
            down = {b for b in range(joy.get_numbuttons()) if joy.get_button(b)}
            ax_i, ay_i = cfg["axes"]
            dz = config.INPUT_CONFIG["axis_deadzone"]
            dx = axis_sign(joy.get_axis(ax_i), dz) if joy.get_numaxes() > ax_i else 0
            soft = (joy.get_axis(ay_i) > dz) if joy.get_numaxes() > ay_i else False
            if dx == 0 and not soft and joy.get_numhats() > cfg["hat_index"]:
                hx, hy = joy.get_hat(cfg["hat_index"])
                dx, soft = hx, hy < 0
        except pygame.error:
            return set(), 0, False
        return down, dx, soft

    def update(self, dt: float) -> None:
        keys = read_keys()
        system = config.INPUT_CONFIG["system"]
        buttons = config.INPUT_CONFIG["buttons"]
        self.keyboard_quit = keyboard_quit_held(keys)
        activity = self.keyboard_quit
        confirm = back = False
        for p in PLAYERS:
            down, dx, soft = self._read_pad(p)
            held_keys = {act for act, codes in self.keymap[p].items() if any(keys[k] for k in codes)}
            if dx == 0:
                dx = ("right" in held_keys) - ("left" in held_keys)
            soft = soft or "down" in held_keys
            pressed = down - self._down[p]
            self._down[p] = down
            new_keys = held_keys - self._keys_prev[p]
            self._keys_prev[p] = held_keys
            cw = bool(pressed & set(buttons["cw"])) or "cw" in new_keys
            ccw = bool(pressed & set(buttons["ccw"])) or "ccw" in new_keys
            self.inputs[p] = PlayerInput(self.repeat[p].update(dt, dx), cw, ccw, soft)
            confirm |= bool(pressed & {*system["confirm_buttons"], system["start_button"]})
            back |= bool(pressed & set(system["back_buttons"]))
            self.start_select[p] = (system["start_button"] in down, system["select_button"] in down)
            if down or dx or soft or held_keys:
                activity = True
        sys_keys = {"confirm"} if any(keys[k] for k in self.confirm_keys) else set()
        sys_keys |= {"back"} if any(keys[k] for k in self.back_keys) else set()
        new = sys_keys - self._sys_prev
        self._sys_prev = sys_keys
        self.confirm = confirm or "confirm" in new
        self.back = back or "back" in new
        self.activity = activity or bool(sys_keys)
        if any(select for _start, select in self.start_select.values()):
            self.select_held_time += dt
        else:
            self.select_held_time = 0.0

    def frame_input(self) -> FrameInput:
        return FrameInput(dict(self.inputs), self.confirm, self.back, self.activity)

    def quit_combo(self) -> bool:
        (s1, sel1), (s2, sel2) = self.start_select["p1"], self.start_select["p2"]
        return should_quit(p1_start=s1, p1_select=sel1, p2_start=s2, p2_select=sel2)
```

```python
"""Synthesised chiptune: sound effects and a looping game tune. No audio
files, no numpy.

The sample generators are adapted from Overquqed's audio.py (same repo).
Music is built by tiling one waveform period per note (array repetition runs
in C), so even the Pi renders a loop quickly, and the tracks are rendered in
a background thread at start-up. If the mixer can't start (no sound card),
everything becomes a silent no-op, so the cabinet never crashes over audio.
"""

from __future__ import annotations

import array
import math
import random
import threading

import pygame

from gate_drop import config
from gate_drop.match import BoardEvent, Cue

SAMPLE_RATE = 22050
AMP = 32767

NOTE_INDEX = {"C": 0, "C#": 1, "D": 2, "D#": 3, "E": 4, "F": 5, "F#": 6, "G": 7, "G#": 8, "A": 9, "A#": 10, "B": 11}


def note_freq(name: str, transpose: int = 0) -> float:
    """'A4' -> 440.0; '-' (rest) -> 0."""
    if name == "-":
        return 0.0
    pitch, octave = name[:-1], int(name[-1])
    n = NOTE_INDEX[pitch] + 12 * (octave + 1) + transpose
    return 440.0 * 2 ** ((n - 69) / 12)


def _wave(kind: str, phase: float) -> float:
    p = phase % 1.0
    if kind == "square":
        return 1.0 if p < 0.5 else -1.0
    if kind == "pulse":  # 25% duty: thinner, NES-like lead
        return 1.0 if p < 0.25 else -1.0
    if kind == "triangle":
        return 4 * p - 1 if p < 0.5 else 3 - 4 * p
    if kind == "saw":
        return 2 * p - 1
    return math.sin(2 * math.pi * p)


def tone(freq: float, dur: float, vol: float, kind: str = "square", decay: float = 0.0,
         slide: float = 0.0, sr: int = SAMPLE_RATE) -> list[int]:
    """A note with optional exponential decay (1/s) and a pitch slide
    (the frequency multiplier reached by the end)."""
    n = max(1, int(sr * dur))
    out = []
    phase = 0.0
    fade = max(1, min(160, n // 6))
    for i in range(n):
        f = freq * (1 + (slide - 1) * i / n) if slide else freq
        phase += f / sr
        env = math.exp(-decay * i / sr) if decay else 1.0
        edge = min(1.0, i / fade, (n - 1 - i) / fade)
        out.append(int(AMP * vol * env * edge * _wave(kind, phase)))
    return out


def noise_wave(dur: float, vol: float, decay: float = 0.0, hold: int = 1, seed: int = 1,
               sr: int = SAMPLE_RATE) -> list[int]:
    """White-ish noise; `hold` repeats each random value (lower = brighter)."""
    rng = random.Random(seed)
    n = max(1, int(sr * dur))
    out = []
    v = 0.0
    for i in range(n):
        if i % hold == 0:
            v = rng.uniform(-1, 1)
        env = math.exp(-decay * i / sr) if decay else 1.0
        out.append(int(AMP * vol * env * min(1.0, (n - 1 - i) / 80) * v))
    return out


def concat(*parts: list[int]) -> list[int]:
    out: list[int] = []
    for p in parts:
        out.extend(p)
    return out


def mix(*tracks: list[int]) -> list[int]:
    n = max(len(t) for t in tracks)
    out = [0] * n
    for t in tracks:
        for i, v in enumerate(t):
            out[i] += v
    return [max(-AMP, min(AMP, v)) for v in out]


def _period(kind: str, freq: float, vol: float, sr: int) -> array.array:
    length = max(2, round(sr / freq))
    return array.array("h", (int(AMP * vol * _wave(kind, i / length)) for i in range(length)))


def voice(notes: list, bpm: float, kind: str, vol: float, staccato: float = 0.85,
          sr: int = SAMPLE_RATE) -> array.array:
    """A melodic line from (note, beats) pairs, built by tiling single periods."""
    out = array.array("h")
    beat = 60.0 / bpm
    for name, beats in notes:
        n = int(sr * beats * beat)
        f = note_freq(name)
        if f <= 0:
            out.extend(array.array("h", [0]) * n)
            continue
        on = int(n * staccato)
        loud, soft = _period(kind, f, vol, sr), _period(kind, f, vol * 0.7, sr)
        attack = min(on, int(sr * 0.04))
        seg = (loud * (attack // len(loud) + 1))[:attack]
        seg.extend((soft * ((on - attack) // len(soft) + 1))[: on - attack])
        out.extend(seg)
        out.extend(array.array("h", [0]) * (n - on))
    return out


def drums(pattern: str, bpm: float, bars: int, vol: float, sr: int = SAMPLE_RATE) -> array.array:
    """8th-note pattern per bar: k = kick, s = snare, h = hat, . = rest."""
    step = int(sr * 60.0 / bpm / 2)
    hits = {
        "k": array.array("h", tone(90, 0.09, vol, "sine", decay=30, slide=0.4)),
        "s": array.array("h", noise_wave(0.08, vol * 0.8, decay=35, hold=2, seed=3)),
        "h": array.array("h", noise_wave(0.025, vol * 0.4, decay=120, hold=1, seed=5)),
    }
    out = array.array("h")
    for _ in range(bars):
        for ch in pattern:
            cell = array.array("h", [0]) * step
            hit = hits.get(ch)
            if hit is not None:
                m = min(step, len(hit))
                cell[:m] = hit[:m]
            out.extend(cell)
    return out

# An original tune: i - VI - III - VII in A minor, 8 bars of eighth notes.

MELODY = [
    "A4 C5 E5 A5 G5 E5 C5 E5", "F4 A4 C5 F5 E5 C5 A4 C5",
    "C5 E5 G5 C6 B5 G5 E5 G5", "G4 B4 D5 G5 F5 D5 B4 D5",
    "A5 G5 E5 C5 E5 G5 A5 -", "F5 E5 C5 A4 C5 E5 F5 -",
    "E5 G5 C6 G5 E5 C5 D5 E5", "D5 B4 G4 B4 D5 G5 B4 -",
]
BASS = ["A2 E3 A2 E3", "F2 C3 F2 C3", "C3 G3 C3 G3", "G2 D3 G2 D3"] * 2
DRUMS = "k.hsk.hs"
TRACKS = {"title": (108, "triangle"), "game": (136, "pulse"), "game:fast": (168, "pulse")}


def render_track(bpm: float, lead: str) -> array.array:
    mel = [(n, 0.5) for bar in MELODY for n in bar.split()]
    bass = [(n, 1.0) for bar in BASS for n in bar.split()]
    a = voice(mel, bpm, lead, 0.22)
    b = voice(bass, bpm, "triangle", 0.30, staccato=0.95)
    d = drums(DRUMS, bpm, len(MELODY), 0.25)
    n = min(len(a), len(b), len(d))
    return array.array("h", (max(-AMP, min(AMP, a[i] + b[i] + d[i])) for i in range(n)))

CLEAR_NOTES = (("C5", "E5"), ("D5", "F#5"), ("E5", "G#5"), ("F#5", "A#5"), ("G#5", "C6"))


def _arp(freqs, dur, vol, kind="square"):
    return concat(*(tone(f, dur, vol, kind, decay=6) for f in freqs))


def build_sfx() -> dict[str, list[int]]:
    c = note_freq
    sfx = {
        "move": tone(520, 0.025, 0.16, "square"),
        "rotate": tone(760, 0.035, 0.2, "pulse", slide=1.3),
        "land": tone(180, 0.06, 0.3, "square", decay=25, slide=0.7),
        "error": concat(_arp([c("E6"), c("G6")], 0.05, 0.25), tone(c("C7"), 0.14, 0.2, "triangle", decay=8)),
        "attack": tone(300, 0.2, 0.3, "saw", slide=3.0),
        "noise": mix(noise_wave(0.18, 0.3, decay=12, hold=3, seed=4),
                     tone(120, 0.18, 0.25, "square", decay=12, slide=0.6)),
        "topout": concat(*(tone(f, 0.12, 0.3, "square", decay=4) for f in (330, 262, 196, 131))),
        "beep": tone(660, 0.08, 0.25, "square"),
        "go": _arp([c("C5"), c("G5"), c("C6")], 0.06, 0.3),
        "round": _arp([c("C5"), c("E5"), c("G5"), c("C6")], 0.08, 0.3),
        "match": _arp([c("C5"), c("E5"), c("G5"), c("C6"), c("G5"), c("C6")], 0.1, 0.3, "triangle"),
    }
    for i, notes in enumerate(CLEAR_NOTES, start=1):
        sfx[f"clear{i}"] = _arp([c(n) for n in notes], 0.05, 0.28)
    return sfx


_CUE_SOUNDS = {"beep": "beep", "go": "go", "round_won": "round", "draw": "round", "match_won": "match"}
_BOARD_SOUNDS = {"move", "rotate", "land", "attack", "noise", "topout"}


def sounds_for(events: list) -> list[str]:
    """The sounds a frame's events ask for, each at most once."""
    names = []
    for e in events:
        if isinstance(e, Cue):
            names.append(_CUE_SOUNDS[e.kind])
        elif isinstance(e, BoardEvent):
            kind = e.event.kind
            if kind in _BOARD_SOUNDS:
                names.append(kind)
            elif kind == "clear":
                names.append(f"clear{min(len(CLEAR_NOTES), max(1, e.event.value))}")
            elif kind == "errors":
                names.append("error")
    return list(dict.fromkeys(names))

class NullAudio:
    def play_events(self, events: list) -> None:
        pass

    def music(self, name: str) -> None:
        pass

    def tempo(self, fast: bool) -> None:
        pass

    def update(self) -> None:
        pass


class Audio:
    def __init__(self) -> None:
        if not pygame.mixer.get_init():
            pygame.mixer.init(SAMPLE_RATE, -16, 1, 512)
        info = pygame.mixer.get_init()
        if not info:
            raise pygame.error("mixer did not initialise")
        self.freq, _fmt, self.channels = info
        pygame.mixer.set_num_channels(16)
        pygame.mixer.set_reserved(1)
        self.music_channel = pygame.mixer.Channel(0)
        self.sounds = {k: self._sound(v) for k, v in build_sfx().items()}
        for s in self.sounds.values():
            s.set_volume(config.MASTER_VOLUME)
        self.tracks: dict[str, pygame.mixer.Sound] = {}
        self.want: str | None = None
        self.fast = False
        self.playing: str | None = None
        self._lock = threading.Lock()
        threading.Thread(target=self._render_all, daemon=True).start()

    def _sound(self, samples) -> pygame.mixer.Sound:
        buf = samples if isinstance(samples, array.array) else array.array("h", samples)
        if self.freq != SAMPLE_RATE:
            ratio = SAMPLE_RATE / self.freq
            buf = array.array("h", (buf[min(len(buf) - 1, int(i * ratio))] for i in range(int(len(buf) / ratio))))
        if self.channels > 1:
            stereo = array.array("h", [0]) * (len(buf) * self.channels)
            for ch in range(self.channels):
                stereo[ch::self.channels] = buf
            buf = stereo
        return pygame.mixer.Sound(buffer=buf.tobytes())

    def _render_all(self) -> None:
        for name, (bpm, lead) in TRACKS.items():
            snd = self._sound(render_track(bpm, lead))
            snd.set_volume(config.MUSIC_VOLUME)
            with self._lock:
                self.tracks[name] = snd

    def play_events(self, events: list) -> None:
        for name in sounds_for(events):
            s = self.sounds.get(name)
            if s is not None:
                s.play()

    def music(self, name: str) -> None:
        if name != self.want:
            self.want = name
            self.update()

    def tempo(self, fast: bool) -> None:
        if fast != self.fast:
            self.fast = fast
            self.update()

    def update(self) -> None:
        """Start (or switch to) the wanted track once it has been rendered."""
        if self.want is None:
            return
        key = self.want + (":fast" if self.fast and f"{self.want}:fast" in TRACKS else "")
        if key == self.playing:
            return
        with self._lock:
            snd = self.tracks.get(key)
        if snd is None:
            return
        self.music_channel.play(snd, loops=-1, fade_ms=150)
        self.playing = key


def make_audio():
    try:
        return Audio()
    except (pygame.error, ValueError, OSError):
        return NullAudio()
```
